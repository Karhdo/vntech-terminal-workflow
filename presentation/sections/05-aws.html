SSO profiles

Log in once, then switch accounts by profile name

# ~/.aws/config (not in the dotfiles repo)
[sso-session loanbud_sso]
sso_start_url = https://<org>.awsapps.com/start
sso_region = us-east-1

[profile loanbud_dev]
sso_session = loanbud_sso
sso_account_id = XXXX
sso_role_name = SpartanPowerUserAccess

[profile loanbud_staging]   # same, other account

[profile loanbud_prod]
sso_session = loanbud_sso
sso_role_name = IAMAuthDatabaseAccess   # …
dailyaws sso login --sso-session loanbud_sso once (browser), then --profile loanbud_dev or export AWS_PROFILE=…. The token is cached and expires: no static keys to rotate.
